--VALORAR HACER ESTE BUCLE POR LOS SUBINFORMES QUE HAYA EN EL INFORME DE SIRBE EN LUGAR DE POR LOS DE REGISTROS (así, para cada subinforme se haría el MERGE de los registros para buscar casillas)
CURSOR cSUBINFORME (
     vpId_fichero NUMBER
     ,vpId_Grupo_Entidad NUMBER
     ,vpINFORME_EXCEL varchar2
     ,vpVersion_excel varchar2
     ,vpFECHA_DATOS_INFORME_GENERADO DATE
 )IS
    SELECT DISTINCT ID_SUBINFORME_SIRBE
    --FROM T_EBE_IMPM_REGISTRO r
    FROM T_EBE_IMPM_AUX_REGISTRO r
    WHERE 
            r.ID_FICHERO = vpId_fichero
        and r.ID_GRUPO_ENTIDAD = vpId_Grupo_Entidad
        and r.ID_INFORME_FICHERO = vpINFORME_EXCEL 
        and r.VERSION_FICHERO= vpVersion_excel
        and r.FECHA_DATOS = vpFECHA_DATOS_INFORME_GENERADO
        and (r.ID_SUBINFORME_SIRBE is not null);

vFILA_CURSOR cSUBINFORME%ROWTYPE;

vSQL VARCHAR2(4000 CHAR);
vAniadirCerosEnExcepciones4410 INT;
vAniadirCerosEnExcepcionesTEBE INT;
vImpEstadoCalculadoOPlantilla INT;



BEGIN
    vPaso := 0;

    vId_Grupo_Entidad := pGrupoEntidad;
	vId_fichero := pIdFichero;
	vFECHA_DATOS_INFORME_GENERADO := pFechaDatos;
	vINFORME_EXCEL := pIdInformeFichero;
	--DECLARE @vINFORME_SIRBE varchar(10) = '3513'  --si un estado no tiene informe sirbe no se puede importar
    vPaso := 5;
	vVersion_excel := pVersionFichero;
    vNUM_CASILLAS_INSERTADAS_EXCEL:=0;
    vPaso := 6;
    vNUM_CASILLAS_INSERTADAS_TOTAL:=0;

	--por defecto sin error
	vNUM_ERRORES := 0;
	vRES_MSG := '';
    
    
    pModoIgnorarCalculadas := NVL(TO_NUMBER(pPermitirAjustarCalculadas), 0);
    --Para El Gestor DE TAREAS
    vHostnameEjecutaTarea := pHostname;
    vTipoGeneracion := 'IMPORTACION';
    
    vID_GENERACION :=
        vID_INFORME_SIRBE_ESCOGIDO || '_' ||
        TO_CHAR(vFECHA_DATOS_INFORME_GENERADO, 'DD/MM/YYYY') || '_' ||
        TO_CHAR(vId_Grupo_Entidad);
    
    vAniadirCerosEnExcepciones4410 := pAniadirCerosEnExcepciones4410;
    vAniadirCerosEnExcepcionesTEBE := pAniadirCerosEnExcepcionesTEBE;

    vImpEstadoCalculadoOPlantilla := pImpEstadoCalculadoOPlantilla;

    IF vDEBUG=1 THEN
        vPaso := 9;
        msgTraza :=  TO_CHAR(SYSDATE, 'YYYYMMDD HH24:MI:SS')|| '.INICIO ' || vProceso 
        || ' . IdFichero: ' || COALESCE(TO_CHAR(pIdFichero),'N/A') 
        || '. Grupo: ' || COALESCE(TO_CHAR(pGrupoEntidad),'N/A') 
        || '. Fecha: ' || COALESCE(TO_CHAR(pFechaDatos),'N/A')  
		|| '. Informe: ' || COALESCE(pIdInformeFichero, 'N/A') 
        || '. Versión: ' || COALESCE(pVersionFichero, 'N/A')
        || '. Versión: ' || COALESCE(pVersionFichero, 'N/A')
        || '. Ajustes: ' || COALESCE(pModoAjustes, 'N/A')
        || '. Sec: ' || COALESCE(pSecuencia_Escogida, -1)
        || '. BorrarAj: ' || COALESCE(pBorrarAjustesPrevios, 'N/A')        
        ;
        DBMS_OUTPUT.put_line (msgTraza);        
        INSERT INTO T_EBE_TRACE (FECHA,CLIENTE,CLAVE2,NIVEL,USUARIO,MENSAJE) VALUES(SYSDATE,'0',SEC_T_EBE_TRACE.NEXTVAL,5,pUsuario,msgTraza);
    END IF;

    ------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
	--Obtener secuencia y versión. anotarlos 
	------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------
	vPaso := 10;
	vID_INFORME_SIRBE_ESCOGIDO := null;
	vINF_SIRBE_ESCOGIDO := null;
	vVERS_SIRBE_ESCOGIDA := null;
	vSEC_INFORME_GENERADO := null;


    
    IF COALESCE(pModoAjustes,0) = 1 THEN
        --en modo ajustes, DEBE existir la pVersion_Escogida y la pSecuencia_Escogida. Si no, saltará la excepción (debe controlarlo la pantalla)
        vPaso := 11;
        SELECT 
			  i.ID_INFORME
			, SUBSTR(i.ID_INFORME,1,4) as INF_SIRBE_ESCOGIDO
			, SUBSTR(i.ID_INFORME,5,6) as VERS_SIRBE_ESCOGIDA
			, gen.SEC_INFORME_GENERADO
             INTO vID_INFORME_SIRBE_ESCOGIDO , vINF_SIRBE_ESCOGIDO, vVERS_SIRBE_ESCOGIDA, vSEC_INFORME_GENERADO
		FROM T_EBE_IMPM_GENERACION impm
		--PTE Tabla de mapeo
		--LEFT JOIN T_EBE_MAPEO_IMPM_SIRBE  
		INNER JOIN  T_EBE_INFORME i on 
			SUBSTR(i.ID_INFORME,1,4) = COALESCE(vINFORME_EXCEL,vINFORME_EXCEL) and i.FECHA_BAJA_INFORME is null
			AND SUBSTR(i.ID_INFORME,5,6) = pVersion_Escogida
		INNER JOIN T_EBE_GENERACION_INFORME gen on gen.ID_GRUPO_ENTIDAD = vId_Grupo_Entidad and gen.ID_INFORME = i.ID_INFORME and gen.SEC_INFORME_GENERADO = pSecuencia_Escogida
			AND gen.FECHA_DATOS_INFORME_GENERADO  = vFECHA_DATOS_INFORME_GENERADO
		WHERE 1=1
			and impm.ID_FICHERO = vId_fichero
			and impm.FECHA_DATOS = vFECHA_DATOS_INFORME_GENERADO
			and impm.ID_GRUPO_ENTIDAD = vId_Grupo_Entidad
			and impm.ID_INFORME_FICHERO = vINFORME_EXCEL
			and impm.VERSION_FICHERO = vVersion_excel;
            
    ELSE
        BEGIN
        vPaso := 12;
        SELECT 
            --@vId_fichero ID_FICHERO, @vId_Grupo_Entidad ID_GRUPO_ENTIDAD, @vFECHA_DATOS_INFORME_GENERADO FECHA_DATOS, @vINFORME_EXCEL INFORME_EXCEL, @vVersion_excel VERSION_EXCEL
            f.ID_INFORME_SIRBE_ESCOGIDO
            , f.INF_ESCOGIDO
            , f.VERS_ESCOGIDA
            , COALESCE(gens.SECUENCIA,1)
            INTO vID_INFORME_SIRBE_ESCOGIDO , vINF_SIRBE_ESCOGIDO, vVERS_SIRBE_ESCOGIDA, vSEC_INFORME_GENERADO
        FROM
        (
            SELECT  
                  CASE WHEN vVersion_excel='DEFAULT'           THEN A.INFORME || A.VERSION_ESCOGIDA
                       WHEN vVersion_excel= A.VERSION_ESCOGIDA THEN A.INFORME || A.VERSION_ESCOGIDA
                       ELSE null   --no se encuentra la versión en SIRBE. No se podrá importar
                    END ID_INFORME_SIRBE_ESCOGIDO
                , A.VERSION_POR_DEFECTO, A.ES_BASICA, A.ES_VIGENTE, A.VERSION_ENCONTRADA
                , A.INFORME INF_ESCOGIDO
                , A.VERSION_ESCOGIDA VERS_ESCOGIDA		
            FROM
            (
                select 
                    SUBSTR(i.ID_INFORME,1,4) INFORME
                --,i.ID_INFORME, i.ID_INFORME CODIGO_INFORME, COALESCE(i.ID_OFICIAL_INFORME,i.ID_ENTIDAD_INFORME) ID_INFORME
                , SUBSTR(i.ID_INFORME,5,6) VERSION_ESCOGIDA
    
                --,i.FECHA_ALTA_INFORME, i.FECHA_BAJA_INFORME,i.FECHA_DESDE, i.FECHA_HASTA ,i.COMENTARIO_VERSION 
                , CASE WHEN i.VERSION_DEFECTO IN ('Si') THEN 1 ELSE 0 END VERSION_POR_DEFECTO
                --, i.ID_TIPO_INFORME, i.ID_CLASE_INFORME, i.ID_SITUACION_INFORME, i.PERIODICIDAD_INFORME, i.FRECUENCIA_INFORME, i.PLAZO_INFORME, i.MES_INICIO_CICLO_INFORME, i.IND_OBLIGATORIO, i.IND_TRATAMIENTO, i.ID_COLECCION, i.ID_MODULO, i.ID_ORDEN, i.ETIQUETAS,i.ENTIDAD_SUPERVISORA,i.COMENTARIO_INFORME
                , CASE WHEN i.COMENTARIO_VERSION IN ('Básica','Basica','Basicas','Básicas') THEN 1 ELSE 0 END ES_BASICA
                , CASE WHEN vFECHA_DATOS_INFORME_GENERADO>=i.FECHA_DESDE AND vFECHA_DATOS_INFORME_GENERADO <=COALESCE(i.FECHA_HASTA,TO_DATE('99991231','YYYYMMDD')) THEN 1 ELSE 0 END ES_VIGENTE
                , CASE WHEN SUBSTR(i.ID_INFORME,5,6) = vVersion_excel THEN 1 ELSE 0 END VERSION_ENCONTRADA
                , ROW_NUMBER() OVER(PARTITION BY SUBSTR(i.ID_INFORME,1,4)
                        ORDER BY SUBSTR(i.ID_INFORME,1,4) ASC
                                , CASE WHEN SUBSTR(i.ID_INFORME,5,6) = vVersion_excel THEN 1 ELSE 0 END DESC
                                , CASE WHEN vFECHA_DATOS_INFORME_GENERADO>=i.FECHA_DESDE AND vFECHA_DATOS_INFORME_GENERADO <=COALESCE(i.FECHA_HASTA,TO_DATE('99991231','YYYYMMDD')) THEN 1 ELSE 0 END DESC
                                , CASE WHEN i.COMENTARIO_VERSION IN ('Básica','Basica','Basicas','Básicas') THEN 1 ELSE 0 END DESC
                                , CASE WHEN i.VERSION_DEFECTO IN ('Si') THEN 1 ELSE 0 END DESC
                                , SUBSTR(i.ID_INFORME,5,6) DESC
                            ) AS seq
    
                from 
                T_EBE_IMPM_GENERACION impm
                --PTE Tabla de mapeo
                --LEFT JOIN T_EBE_MAPEO_IMPM_SIRBE  
                INNER JOIN  T_EBE_INFORME i on 
                    SUBSTR(i.ID_INFORME,1,4) = COALESCE(vINFORME_EXCEL,vINFORME_EXCEL) and i.FECHA_BAJA_INFORME is null
    
                WHERE 1=1
                    and impm.ID_FICHERO = vId_fichero
                    and impm.FECHA_DATOS = vFECHA_DATOS_INFORME_GENERADO
                    and impm.ID_GRUPO_ENTIDAD = vId_Grupo_Entidad
                    and impm.ID_INFORME_FICHERO = vINFORME_EXCEL
                    and impm.VERSION_FICHERO = vVersion_excel
            ) A
            WHERE 1=1
            AND A.seq = 1
        )  f
        LEFT JOIN (
            select 
                ID_GRUPO_ENTIDAD, ID_INFORME, FECHA_DATOS_INFORME_GENERADO
                ,SUM(1) as N	
                --si hay alguna generada eb estado generado, esa. si no, creamos una nueva. nunca sobreescribimos una en análisis,validada,enviada,aceptada,descartada,rechazada
                , CASE WHEN MAX(CASE WHEN ID_SITUACION_INFORME_GENERADO IN ('0001','0001E','0001C') THEN SEC_INFORME_GENERADO ELSE -1 END) = -1 THEN COALESCE(MAX(SEC_INFORME_GENERADO),0) + 1 
                       ELSE MAX(CASE WHEN ID_SITUACION_INFORME_GENERADO IN ('0001','0001E','0001C') THEN SEC_INFORME_GENERADO ELSE -1 END) 
                  END SECUENCIA
    
                , MAX(CASE WHEN ID_SITUACION_INFORME_GENERADO IN ('0001','0001E','0001C') THEN 1 ELSE 0 END) SOBREESCRIBIR_GENERACION
    
            from T_EBE_GENERACION_INFORME gen
            group by ID_GRUPO_ENTIDAD, ID_INFORME, FECHA_DATOS_INFORME_GENERADO
        ) gens on gens.ID_GRUPO_ENTIDAD = vId_Grupo_Entidad and gens.ID_INFORME = f.ID_INFORME_SIRBE_ESCOGIDO and gens.FECHA_DATOS_INFORME_GENERADO = vFECHA_DATOS_INFORME_GENERADO;
        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                vID_INFORME_SIRBE_ESCOGIDO := null;
                vINF_SIRBE_ESCOGIDO := null;
                vVERS_SIRBE_ESCOGIDA := null;
                vSEC_INFORME_GENERADO := null;
        END;
    END IF;


	IF vDEBUG=1 THEN
		msgTraza :=  TO_CHAR(SYSDATE, 'YYYYMMDD HH24:MI:SS')|| '.Id.Informe.SIRBE: ' || COALESCE(vID_INFORME_SIRBE_ESCOGIDO,'N/A') || '. Secuencia: ' || COALESCE(TO_CHAR(vSEC_INFORME_GENERADO),'N/A');
		DBMS_OUTPUT.put_line (msgTraza);
        INSERT INTO T_EBE_TRACE (FECHA,CLIENTE,CLAVE2,NIVEL,USUARIO,MENSAJE) VALUES(SYSDATE,'0',SEC_T_EBE_TRACE.NEXTVAL,5,pUsuario,msgTraza);
    END IF;

	IF ((vID_INFORME_SIRBE_ESCOGIDO IS null) OR (vSEC_INFORME_GENERADO IS null)) THEN
		vPaso := 15;
		vErrorMessage := 'No se encuentra el informe a importar o una secuencia disponible. Informe Excel: ' || COALESCE(vINFORME_EXCEL,'N/A');

        IF vDEBUG=1 THEN

                --  porque en la tabla VL se agrupó por ellos!!

                        --, CASE WHEN COALESCE(VL.ID_VARIACION_FILA,'-1')='-1' THEN 'ZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZZ' else VL.ID_VARIACION_FILA END COLLATE Modern_Spanish_CI_AI --SQL-Server

                ) AS ORDEN_VARIACION_LOCAL

                        AND COALESCE(v.ID_VARIACION_1,'-1') = COALESCE(a.ID_VARIACION_1,'-1') AND COALESCE(v.ID_VARIACION_2,'-1') = COALESCE(a.ID_VARIACION_2,'-1')
                GROUP BY a.ID_VARIACION_1, a.ID_VARIACION_2

            --tabla con las variaciones locales de cada variación global

                                where ID_FICHERO=vId_fichero and FECHA_DATOS =vFECHA_DATOS_INFORME_GENERADO and ID_GRUPO_ENTIDAD=vId_Grupo_Entidad and ID_INFORME_FICHERO=vINFORME_EXCEL and VERSION_FICHERO=vVersion_excel
