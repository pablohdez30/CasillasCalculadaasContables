					, a.ID_CASILLA
					, a.TIPO_ORIGEN
					, a.VARIACIONES_APLICA
			),
			CALCULADAS_A_IGNORAR AS
			(
				SELECT
					  x.ID_CASILLA
					, x.ID_VARIACION_1
					, x.ID_VARIACION_2
					, x.ID_VARIACION_FILA
					, CASE
						WHEN COUNT(o.ID_CASILLA) = 0 THEN 0

						-- Si existe algún origen no calculado/sin variaciones_aplica => NO calculada
						WHEN MAX(CASE WHEN o.TIPO_ORIGEN NOT IN (1,4) AND o.VARIACIONES_APLICA_NORM IS NULL THEN 1 ELSE 0 END) = 1 THEN 0

						-- Modo 4: si hay origen externo sin variaciones_aplica => NO calculada
						WHEN COALESCE(@pModoIgnorarCalculadas,0) = 4
							 AND MAX(CASE WHEN o.ES_MISMO_INFORME = 0 AND o.VARIACIONES_APLICA_NORM IS NULL THEN 1 ELSE 0 END) = 1 THEN 0

						-- Si todas las calculadas con aplica NO aplican a esta variación => NO calculada
						WHEN
							SUM(CASE WHEN o.TIPO_ORIGEN = 1 AND o.VARIACIONES_APLICA_NORM IS NOT NULL THEN 1 ELSE 0 END) > 0
							AND
							SUM(
								CASE
									WHEN o.TIPO_ORIGEN = 1
									 AND o.VARIACIONES_APLICA_NORM IS NOT NULL
									 AND (
											-- SIN '$' => aplica solo a VG1
											(
												p.VG2_LIST IS NULL
												AND p.VG1_LIST IS NOT NULL
												AND CHARINDEX(',' + x.ID_VARIACION_1 + ',', ',' + p.VG1_LIST + ',') = 0
											)
											OR
											-- CON '$' => aplica a VG1 y VG2 (si no hay '&')
											(
												p.VG2_LIST IS NOT NULL
												AND p.VG1_LIST IS NOT NULL
												AND CHARINDEX(',' + x.ID_VARIACION_1 + ',', ',' + p.VG1_LIST + ',') > 0
												AND CHARINDEX(',' + x.ID_VARIACION_2 + ',', ',' + p.VG2_LIST + ',') = 0
												AND CHARINDEX('&', p.VG2_LIST) = 0
											)
										 )
									THEN 1 ELSE 0
								END
							)
							=
							SUM(CASE WHEN o.TIPO_ORIGEN = 1 AND o.VARIACIONES_APLICA_NORM IS NOT NULL THEN 1 ELSE 0 END)
						THEN 0

						ELSE 1
					  END AS ES_CALCULADA
				FROM #T_EBE_IMPM_AUX_CASILLAS_VAR x
				LEFT JOIN ORIG o
					ON  o.ID_SUBINFORME = @vID_SUBINFORME
					AND o.ID_CASILLA    = x.ID_CASILLA
				OUTER APPLY
				(
					SELECT
						CASE
							WHEN o.VARIACIONES_APLICA_NORM IS NULL THEN NULL
							WHEN CHARINDEX('$', o.VARIACIONES_APLICA_NORM) > 0
								THEN LEFT(o.VARIACIONES_APLICA_NORM, CHARINDEX('$', o.VARIACIONES_APLICA_NORM) - 1)
							ELSE o.VARIACIONES_APLICA_NORM
						END AS VG1_LIST,
						CASE
							WHEN o.VARIACIONES_APLICA_NORM IS NULL THEN NULL
							WHEN CHARINDEX('$', o.VARIACIONES_APLICA_NORM) > 0
								THEN SUBSTRING(o.VARIACIONES_APLICA_NORM, CHARINDEX('$', o.VARIACIONES_APLICA_NORM) + 1, 4000)
							ELSE NULL
						END AS VG2_LIST
				) p
				WHERE x.SITUACION = 'VIENE'
				GROUP BY
					  x.ID_CASILLA
					, x.ID_VARIACION_1
					, x.ID_VARIACION_2
					, x.ID_VARIACION_FILA
			)
			UPDATE x
			   SET x.SITUACION = 'IGNORADA' -- IGNORADA = 'VIENE' pero quiero ignorarla
			FROM #T_EBE_IMPM_AUX_CASILLAS_VAR x
			INNER JOIN CALCULADAS_A_IGNORAR c
				ON  c.ID_CASILLA        = x.ID_CASILLA
				AND c.ID_VARIACION_1    = x.ID_VARIACION_1
				AND c.ID_VARIACION_2    = x.ID_VARIACION_2
				AND c.ID_VARIACION_FILA = x.ID_VARIACION_FILA
			WHERE x.SITUACION = 'VIENE'-- DUPLICADO, RGI, NO VIENE
			  AND c.ES_CALCULADA = 1;

			SET @vNFA = @@ROWCOUNT;
			IF (@vDEBUG = 1)
			BEGIN
				SET @msgTraza = 'UPDATE #T_EBE_IMPM_AUX_CASILLAS_VAR -> IGNORADA (calculadas): ' + CAST(@vNFA AS varchar);
				PRINT CONVERT(varchar, GETDATE(), 113) + '. ' + @msgTraza;
			END;
			
		END;
